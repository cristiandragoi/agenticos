    99	def ensure_layout() -> list:

   134	def iso_utc(moment=None) -> str:
   135	    moment = moment or now_utc()
   136	    if moment.tzinfo is None:
   137	        moment = moment.replace(tzinfo=timezone.utc)
   138	    return moment.astimezone(timezone.utc).strftime(ISO_FORMAT)

   141	def parse_iso(value):
   142	    """Parse an ISO-8601 'Z' timestamp; return None when unparsable."""
   143	    if not isinstance(value, str) or not value:
   144	        return None
   145	    try:
   146	        return datetime.strptime(value, ISO_FORMAT).replace(tzinfo=timezone.utc)
   147	    except ValueError:
   148	        return None

   154	def read_json(path, default=None):
   155	    """Read JSON, returning *default* when the file is absent or corrupt.
   156	
   157	    A corrupt state file must never raise at import/startup time -- the routine
   158	    has to be able to run and report the problem.
   159	    """
   160	    p = Path(path)
   161	    try:
   162	        with open(p, "r", encoding="utf-8") as fh:
   163	            return json.load(fh)
   164	    except (OSError, ValueError):
   165	        return default
   166

   168	def write_json_atomic(path, obj) -> Path:
   169	    """Write JSON via a unique temp file + os.replace (never a partial file)."""
   170	    p = Path(path)
   171	    p.parent.mkdir(parents=True, exist_ok=True)
   172	    tmp = p.parent / (".%s.%d.%s.tmp" % (p.name, os.getpid(), uuid.uuid4().hex))
   173	    payload = json.dumps(obj, indent=2, sort_keys=False, ensure_ascii=False)
   174	    with open(tmp, "w", encoding="utf-8", newline="\n") as fh:
   175	        fh.write(payload)
   176	        fh.write("\n")
   177	        fh.flush()
   178	        os.fsync(fh.fileno())
   179	    os.replace(tmp, p)
   180	    return p

   182	
   183	def append_jsonl(path, obj) -> Path:
   184	    """Append one JSON object as a single line, using one O_APPEND write.
   185	
   186	    Append mode means the file is never rewritten, so an existing line can
   187	    never be lost -- the alert log is the routine's evidence record.
   188	    """
   189	    p = Path(path)
   190	    p.parent.mkdir(parents=True, exist_ok=True)
   191	    line = json.dumps(obj, sort_keys=True, ensure_ascii=False) + "\n"
   192	    blob = line.encode("utf-8")
   193	    fd = os.open(str(p), os.O_APPEND | os.O_CREAT | os.O_WRONLY, 0o644)
   194	    try:
   195	        written = os.write(fd, blob)
   196	        if written != len(blob):
   197	            raise OSError("short append to %s (%d of %d bytes)" % (p, written, len(blob)))
   198	    finally:
   199	        os.close(fd)
   200	    return p
   201	
   202	
   203	def read_jsonl(path) -> list:
   204	    """Read a JSONL file, skipping unparsable lines."""
   205	    out = []
   206	    p = Path(path)
   207	    if not p.exists():
   208	        return out
   209	    with open(p, "r", encoding="utf-8") as fh:
   210	        for raw in fh:
   211	            raw = raw.strip()
   212	            if not raw:
   213	                continue
   214	            try:
   215	                out.append(json.loads(raw))
   216	            except ValueError:
   217	                continue
   218	    return out
